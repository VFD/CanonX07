### jeu de la vie


```basic
10 CLEAR 50,&H9F0: A=&HA00
20 PRINT HEX$(A)
30 S=0
40 FOR I=1 TO 8
50 LINE INPUT A$
60 V=VAL("&H"+A$)
70 POKEA,J
80 A=A+1: S=S+V
90 NEXT
100 INPUT"SOMME ",SL
110 IF SL=S THEN 20
120 BEEP 20,2: PRINT "ERREUR"
130 A=A-8: GOTO 20
```

```basic

20 INIT#1,"CASI:"
30 INPUT#1,N$,D,F
40 MOTOR
50 PRINT"TROUVE : ";N$
60 FOR I=D-1 TO F
70 POKE I,INP(#1)
80 NEXT: MOTOR OFF
90 END

120 N$="VIE": INIT#1,"CASO:"

140 PRINT#1,N$,D,F: MOTOR
150 FOR I=1 TO 1800: NEXT
160 FOR I=D TO F: OUT#I,PEEK(I)
170 NEXT: MOTOR OFF
180 END
```
